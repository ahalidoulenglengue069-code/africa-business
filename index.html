async function saveMarketProduct(){ if(!marketCanPublish()){ alert('Seul le fondateur peut publier dans le Marché.'); return; } const btn=document.getElementById('publishMarketBtn'); const cancel=document.getElementById('cancelMarketBtn'); if(btn){ btn.disabled=true; btn.innerHTML='⏳ VÉRIFICATION DU COMPTE.....'; btn.style.opacity='0.7'; } if(cancel) cancel.disabled=true; try{ const f=document.getElementById('mpPhoto')?.files?.[0]; const name=document.getElementById('mpName')?.value.trim(); const price=Number(document.getElementById('mpPrice')?.value||0); const desc=document.getElementById('mpDesc')?.value.trim(); if(!name || price<=0){ throw new Error('Nom et prix valides obligatoires.'); } if(!state.userId || !state.token){ throw new Error('Session Fondateur absente. Déconnecte-toi puis reconnecte-toi.'); } /* Récupération directe du profil Fondateur */ const profiles=await api( `/rest/v1/profils_busaf?user_id=eq.${encodeURIComponent(state.userId)}&select=id,user_id,nom_complet,role,Fondateur&limit=1` ); const founderProfile=Array.isArray(profiles)?profiles[0]:profiles; if(!founderProfile?.id){ throw new Error( 'Profil Fondateur introuvable dans profils_busaf pour ce compte.' ); } const isFounderProfile= String(founderProfile.role||'').trim().toLowerCase()==='fondateur' || ['true','oui','yes','fondateur'].includes( String(founderProfile.Fondateur||'').trim().toLowerCase() ); if(!isFounderProfile){ throw new Error( 'Ce compte est connecté mais son profil Supabase n’est pas reconnu comme Fondateur.' ); } if(btn) btn.innerHTML='⏳ VÉRIFICATION DE LA PHOTO.....'; let image_url=null; if(f){ if(!f.type.startsWith('image/')){ throw new Error('Le fichier doit être une image.'); } if(f.size>5*1024*1024){ throw new Error('Image trop lourde : maximum 5 Mo.'); } const ext= (f.name.split('.').pop()||'jpg') .toLowerCase() .replace(/[^a-z0-9]/g,'')||'jpg'; const path= `marche/${state.userId}/product-${Date.now()}.${ext}`; if(btn) btn.innerHTML='⏳ ENVOI DE LA PHOTO.....'; const r=await fetch( `${SUPABASE_URL}/storage/v1/object/profils/${path}`, { method:'POST', headers:{ apikey:SUPABASE_KEY, Authorization:'Bearer '+state.token, 'Content-Type':f.type, 'x-upsert':'true' }, body:f } ); const txt=await r.text(); if(!r.ok){ throw new Error( `Photo refusée par Supabase (${r.status}) : ${txt||'AccessDenied'}` ); } image_url= `${SUPABASE_URL}/storage/v1/object/public/profils/${path}`; } if(btn) btn.innerHTML='⏳ ENREGISTREMENT DANS SUPABASE.....'; const body={ nom:name, description:desc||null, prix:price, type_produit:'digital', createur_id:founderProfile.id, image_url:image_url }; const rows=await api( '/rest/v1/produits?select=id,nom,description,prix,type_produit,createur_id,created_at,image_url', { method:'POST', headers:{ 'Prefer':'return=representation' }, body:JSON.stringify(body) } ); const created=Array.isArray(rows)?rows[0]:rows; if(!created?.id){ throw new Error( 'Supabase n’a pas retourné le produit créé.' ); } if(btn) btn.innerHTML='⏳ ACTUALISATION DU MARCHÉ.....'; state.marketCatalogLoaded=false; await loadMarketCatalog(true); closeModal(); render(); openModal( '✅ Produit publié', `
Le produit ${esc(created.nom||name)} a été enregistré avec succès dans le Marché BuSAF.
` ); }catch(e){ console.error('Erreur ajout produit BuSAF:',e); if(btn){ btn.disabled=false; btn.innerHTML='VALIDER ET PUBLIER'; btn.style.opacity='1'; } if(cancel) cancel.disabled=false; alert( '❌ AJOUT DU PRODUIT IMPOSSIBLE\n\n'+ (e?.message||'Erreur inconnue') ); } }
